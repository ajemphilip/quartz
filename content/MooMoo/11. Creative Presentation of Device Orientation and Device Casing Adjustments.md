# Goals
- Create modular system to attach printed items to front and back of the device
- Create head and tails shapes to creatively indicate device orientation

# Description
The design proposal introduces the modification of the braille device with additional functionality that will serve to identify front and back of the device. Its was identified, since braille pin is symmetrical, to indicate front and back to avoid confusion and teach braille letters with clarity for the users. 

# Prototyping
![[PENUP_20250425_200547.jpg]]
Head and Tail Sketch

The modifications changed only top tray to have hole in front and at the back of the tray to accommodate modular attachment of the head. Due to children interest, generally low difficulty of the creation and to match the device theme, the device imitate the cow and the head shape is formed as rectangular. The tail is also simple curved line significantly smaller in size to emphasize in the top and bottom difference.

// ============================
// Curved Multi-Hole Panel + Holder Holes
// ============================

$fn = 100; // global smoothness

// ---- Top surface (Minkowski) ----
TOP_CORE_SIZE   = [185,120,100];
TOP_CORE_Z      = 100;
KERNEL_R        = 10;

// ---- Frame plate (inner frame subtraction) ----
FRAME_OUT       = [20,120,5];
FRAME_IN        = [40,115,10];
FRAME_Z         = 31;

// ---- Pin holes (6 total) ----
HOLE_H          = 35;
HOLE_R          = 22;
HOLE_Z          = 15;
HOLE_POS        = [
  [  0,  32.5],
  [  0, -32.5],
  [ -65,  32.5],
  [ -65, -32.5],
  [  65,  32.5],
  [  65, -32.5]
];

// ---- Hole supports (rings at top) ----
SUP_Z           = 152.4;
SUP_OUT_H       = 15;
SUP_OUT_R       = 22.5;
SUP_IN_H        = 16;
SUP_IN_R        = 20.5;

// ---- Grip dots ----
DOT_R           = 1;
DOT_POS         = [
  // side dots
  [ 99,   0, 125],
  [-99,   0, 125],
  // other dots
  [  0, 65.1, 125],
  [  0,-65.1, 125],
  // attachment cube (right)
  [ 99,   0, 143.5],
  [ 99,   0, 133],
  [ 99,   5, 138],
  [ 99,  -5, 138],
  // attachment cube (left)
  [-99,   0, 143.5],
  [-99,   0, 133],
  [-99,   5, 138],
  [-99,  -5, 138]
];

// ---- Shape holder holes (front/back cubes) ----
HOLDER_CUBE_EDGE = 10;
FRONT_HOLDER_POS = [-98, 0, 138];
BACK_HOLDER_POS  = [ 98, 0, 138];

// ---- Large stacked rectangle diff block ----
STACK_Z         = 80;
STACK_OUT       = [215,142,100];
STACK_RING_Z    = 46;
STACK_RING_OUT  = [215,151,10];
STACK_RING_IN   = [198,131,10];

// ---- Cleanup carves ----
CUT_TOP_Z       = 84;
CUT_TOP_SIZE    = [191,125,141];
CUT_BOTTOM_Z    = -22;
CUT_BOTTOM_SIZE = [300,300,105];

// ==================== Modules ====================

module topSurface() {
  minkowski() {
    translate([0,0,TOP_CORE_Z]) cube(TOP_CORE_SIZE, center=true);
    sphere(r=KERNEL_R, $fn=20);
  }
}

module framePlate() {
  difference() {
    cube(FRAME_OUT, center=true);
    cube(FRAME_IN,  center=true);
  }
}

module holes() {
  for (p = HOLE_POS)
    translate([p[0], p[1], HOLE_Z])
      cylinder(h=HOLE_H, r=HOLE_R, center=true);
}

module holeSupport() {
  for (p = HOLE_POS)
    difference() {
      translate([p[0], p[1], SUP_Z])
        cylinder(h=SUP_OUT_H, r=SUP_OUT_R, center=true);
      translate([p[0], p[1], SUP_Z])
        cylinder(h=SUP_IN_H,  r=SUP_IN_R,  center=true);
    }
}

module gripDots() {
  for (p = DOT_POS)
    translate(p) sphere(r=DOT_R);
}

module front_shape_holder_hole() {
  translate(FRONT_HOLDER_POS) cube(HOLDER_CUBE_EDGE, center=true);
}

module back_shape_holder_hole() {
  translate(BACK_HOLDER_POS) cube(HOLDER_CUBE_EDGE, center=true);
}

module differenceCube() {
  translate([0,0,STACK_Z])
    difference() {
      cube(STACK_OUT, center=true);
      translate([0,0,STACK_RING_Z])
        difference() {
          cube(STACK_RING_OUT, center=true);
          cube(STACK_RING_IN,  center=true);
        }
    }
}

// ==================== Assembly ====================

module curved_multi_hole_panel_with_holders() {
  union() {
    gripDots();
    holeSupport();

    minkowski() {
      difference() {
        topSurface();
        translate([0,0,FRAME_Z]) framePlate();
        differenceCube();
        translate([0,0,150]) holes();
        translate([0,0, CUT_TOP_Z])    cube(CUT_TOP_SIZE,    center=true);
        translate([0,0, CUT_BOTTOM_Z]) cube(CUT_BOTTOM_SIZE, center=true);
        front_shape_holder_hole();
        back_shape_holder_hole();
      }
    }
  }
}

// ---------- Entry ----------
curved_multi_hole_panel_with_holders();
```

// ============================
// Head + Features (template-organized)
// ============================

// ==================== Modules ====================

module head() {
  cube([60,80,70], center=true);
}

module cubeHolder() {
  translate([0,-1,0])
    cube([10.8,13,25], center=true);
}

module supportBar() {
  translate([0,7,11])
    cube([5,4,3], center=true);
}

module mouth() {
  // top mouth
  translate([38,0,-5])
    cube([1,30,1], center=true);

  // side mouth - left
  translate([38,-25,-20])
    rotate([90,0,0])
      cube([1,15,1], center=true);

  // side mouth - right
  translate([38,25,-20])
    rotate([90,0,0])
      cube([1,15,1], center=true);

  // main mouth
  translate([45,0,-20])
    cube([10,35,18], center=true);
}

module eyes() {
  // right eye
  translate([40,25,15])
    cube([5,30,20], center=true);
  translate([40,30,15])
    sphere(6);

  // left eye
  translate([40,-25,15])
    cube([5,30,20], center=true);
  translate([40,-30,15])
    sphere(6);
}

module nostrils() {
  // left nostril
  translate([59,8,-19])
    cube([3,10,10]);

  // right nostril
  translate([59,-16,-19])
    cube([3,10,10]);
}

module ear() {
  translate([100,10,10])
    cylinder(20,10,5, center=true);
  rotate([180,0,0])
    translate([100,-10,10])
      cylinder(20,10,5, center=true);
}

// ==================== Assembly ====================

module head_with_features() {
  difference() {
    minkowski() {
      head();
      sphere(10, $fn=20);
    }
    translate([-40,0,5])
      rotate([90,0,90]) {
        cubeHolder();
        supportBar();
      }
  }

  minkowski() {
    mouth();
    sphere(10, $fn=20);
  }

  eyes();

  translate([-100,50,0])
    rotate([45,0,0])
      ear();

  translate([-100,-65,10])
    rotate([-45,0,0])
      ear();

  translate([20,30,50])
    cylinder(30,10,5, center=true);

  translate([20,-30,50])
    cylinder(30,10,5, center=true);

  nostrils();
}

// ---------- Entry ----------
head_with_features();

## Tail
### Image
![[Pasted image 20250411211837.png]]
### Code

// ============================
// Bezier Hull Curve + Tail & Mount (template-organized)
// ============================

// Taken From: https://raphaelluckom.com/posts/bezier_curves.html

// ---- Parameters ----
SPHERE_POINTS = [
  [-1,  0, -20],
  [ 0, 20,   0],
  [ 3, 40,   0],
  [ 7, 50,   0]
];
SPHERE_R = 3;

MOUNT_SIZE = [9.5, 9.5, 10];
MOUNT_POS  = [0, -12, -10];

// ==================== Modules ====================

module piecewise_join(points) {
  for (n=[1:len(points) - 2]) {
    hull() {
      translate(points[n-1]) children(0);
      translate(points[n])   children(0);
    }
    hull() {
      translate(points[n+1]) children(0);
      translate(points[n])   children(0);
    }
  }
}

module mount() {
  cube(MOUNT_SIZE, center=true);
}

module tailEnd() {
  translate([-93,75,-10])
  rotate([90,0,0]) {
    translate([100,10,10])
      cylinder(30,10,5, center=true);
    rotate([180,0,0])
    translate([100,-10,20])
      cylinder(30,10,5, center=true);
  }
}

// ==================== Assembly ====================

module curve_tail_mount() {
  piecewise_join(SPHERE_POINTS) sphere(SPHERE_R);
  tailEnd();
  translate(MOUNT_POS) mount();
}

// ---------- Entry ----------
curve_tail_mount();
```

# Critical Reflection
The shapes attach to the holder and tray very well. Despite they move and feel a bit unstable when the device is moved they offer apparent difference between back and front. The shapes and details are well articulated haptically making them feelable by touch.
