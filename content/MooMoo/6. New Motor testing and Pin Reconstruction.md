# Goals
- Adjust the pin module to meet the needs of new motor
- Test motor 50:1 motor power (torque) in terms of screw friction block and speed of moving pin up and down

# Description 
The regular gearless dc motor was unable to move pin when at maximum or minimum spot because of the metal friction between the nut and screw. Unfortunately all means of solving the situation through means other then motor replacement failed. Because of that, the old dc motors was replaced with geared motor powerful motor. Since the new motor is significantly smaller some adjustments had to be done to the pin casing. When finished the whole system was tested and its functionality reflected
# Prototyping

## Bottom Pin
Because of the new motor, top pin was reconstructed to fit the motor component. Because of significantly smaller size of the motor the pin casing is thinner and possibly will benefit future space requirement for other electronic boards. Since the motor is rectangular the casing also mimics the shape of the motor. Because of that, the requirements for railings is not necessary to keep the top pin in place.

// ==========================
// Small Enclosure (Parametric)
// ==========================

$fn = 64;  // smoothness for spheres/cyls

// ---- Main body ----
MAIN_OUT      = [17, 17, 25];
MAIN_IN       = [14, 12, 26];
MAIN_IN_ZOFF  = 3;              // inner cut shift

// ---- Cable holes (two) ----
CABLE_H       = 10;
CABLE_R       = 3;
CABLE_Z       = -8;
CABLE_X_OFF   = 4.5;

// ---- Bottom attachment cube ----
ATTACH_SIZE   = [8, 8, 6];
ATTACH_Z      = -14;

// ---- Grip dots (4 spheres) ----
DOT_R         = 1;
DOT_POS       = [ [ 0,  5.6, 0],
                  [ 0, -5.6, 0],
                  [ 6.6, 0,  0],
                  [-6.6, 0,  0] ];

// ==================== Modules ====================

module main_body() {
  difference() {
    cube(MAIN_OUT, center=true);
    translate([0,0,MAIN_IN_ZOFF])
      cube(MAIN_IN, center=true);
  }
}

module cable_holes() {
  for (x = [-CABLE_X_OFF, CABLE_X_OFF])
    translate([x, 0, CABLE_Z])
      cylinder(h=CABLE_H, r=CABLE_R, center=true);
}

module bottom_attachment() {
  translate([0,0,ATTACH_Z])
    cube(ATTACH_SIZE, center=true);
}

module grip_dots() {
  for (p = DOT_POS)
    translate(p) sphere(r=DOT_R);
}

module enclosure() {
  // main body with cable holes removed
  difference() {
    main_body();
    cable_holes();
  }

  // add-on solids
  bottom_attachment();
  grip_dots();
}

// ---------- Entry ----------
enclosure();
```

// ===========================
// Rail Pin Casing Assembly
// ===========================

$fn = 100; // global smoothness

// ---- Rail pin casing (bottom shell) ----
CASING_OUT_SIZE   = [21, 21, 52];
CASING_OUT_Z      = -23;
CASING_IN_SIZE    = [18, 18, 48];
CASING_IN_Z       = -28;

// ---- Transfer block (top chunk) ----
XFER_SIZE         = [14, 15, 33];
XFER_Z            = 15.5;

// ---- Grip dots ----
DOT_R             = 1;
DOT_Z             = 35;
DOT_OFFSETS       = [[ 0,  6.8], [ 0, -6.8], [ 6.4, 0], [-6.4, 0]];

// ---- Tiny screw guide dots ----
TINY_DOT_R        = 0.2;
TINY_DOT_Z        = -3;
TINY_DOT_OFFSETS  = [[0,  5], [0, -5]];

// ---- Screw holder (hex) & screw clearance ----
HOLDER_H          = 6.5;
HOLDER_R          = 5.8;  // circumscribed hex radius
HOLDER_Z          = -4;

CLEAR_H           = 63;
CLEAR_R           = 4.5;

// ---- Top button holder block ----
TOP_HOLDER_OUT    = [17.5, 18.5, 13.5];
TOP_HOLDER_OUT_Z  = 34;
TOP_HOLDER_IN     = [12.6, 13.6, 20];
TOP_HOLDER_IN_Z   = 31;

// ==================== Modules ====================

module rail_pin_casing() {
  difference() {
    translate([0,0,CASING_OUT_Z]) cube(CASING_OUT_SIZE, center=true);
    translate([0,0,CASING_IN_Z])  cube(CASING_IN_SIZE,  center=true);
  }
}

module transfer_block() {
  translate([0,0,XFER_Z]) cube(XFER_SIZE, center=true);
}

module screw_holder_hex() {
  translate([0,0,HOLDER_Z])
    cylinder(h=HOLDER_H, r=HOLDER_R, $fn=6, center=true);
}

module screw_clearance() {
  cylinder(h=CLEAR_H, r=CLEAR_R, center=true);
}

module top_button_holder() {
  difference() {
    translate([0,0,TOP_HOLDER_OUT_Z]) cube(TOP_HOLDER_OUT, center=true);
    translate([0,0,TOP_HOLDER_IN_Z])  cube(TOP_HOLDER_IN,  center=true);
  }
}

module grip_dots() {
  for (p = DOT_OFFSETS)
    translate([p[0], p[1], DOT_Z]) sphere(r=DOT_R);
}

module grip_dots_screw() {
  for (p = TINY_DOT_OFFSETS)
    translate([p[0], 0, TINY_DOT_Z]) sphere(r=TINY_DOT_R);
}

module differenced_top() {
  difference() {
    transfer_block();
    screw_holder_hex();
    screw_clearance();
  }
}

module differenced_bottom() {
  difference() {
    rail_pin_casing();
    screw_holder_hex();
    screw_clearance();
  }
}

module rail_pin_casing_assembly() {
  union() {
    differenced_bottom();
    differenced_top();
    top_button_holder();
    grip_dots();
    grip_dots_screw();
  }
}

// ---------- Entry ----------
rail_pin_casing_assembly();
```

## Electronic Components
The pin system used Arduino UNO microcontroller to connect and control all the parts.
### Code
```
// ===================================
// Single Button → Motor Toggle Driver
// ===================================

// ---- Button ----
const int BUTTON_PIN = 9;

// ---- Motor Driver Pins (Motor A) ----
const int PWM_A = 4;
const int IN1_A = 5;
const int IN2_A = 3;

// ---- Behavior Constants ----
const unsigned long MOTOR_RUN_TIME = 2000; // ms
const int MOTOR_SPEED = 255;
const unsigned long DEBOUNCE_DELAY = 50;   // ms for stable button read

// ---- State Variables ----
bool motorExtended = false;        // 0 = retracted, 1 = extended
bool lastButtonState = HIGH;       // pulled high by internal resistor
unsigned long lastDebounceTime = 0;

void setup() {
  Serial.begin(9600);

  // Button input with internal pull-up
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  // Motor control pins
  pinMode(PWM_A, OUTPUT);
  pinMode(IN1_A, OUTPUT);
  pinMode(IN2_A, OUTPUT);

  digitalWrite(PWM_A, 0);
}

void loop() {

  bool reading = digitalRead(BUTTON_PIN);

  // ---- Debounce ----
  if (reading != lastButtonState) {
    lastDebounceTime = millis();
  }

  if ((millis() - lastDebounceTime) > DEBOUNCE_DELAY) {
    // Button pressed (LOW → GND)
    if (reading == LOW && lastButtonState == HIGH) {
      toggleMotor();
    }
  }

  lastButtonState = reading;
}

// ===================================
// Toggle motor direction + move
// ===================================
void toggleMotor() {

  if (!motorExtended) {
    Serial.println("EXTENDING...");
    digitalWrite(IN1_A, LOW);
    digitalWrite(IN2_A, HIGH);
  } else {
    Serial.println("RETRACTING...");
    digitalWrite(IN1_A, HIGH);
    digitalWrite(IN2_A, LOW);
  }

  digitalWrite(PWM_A, MOTOR_SPEED);
  delay(MOTOR_RUN_TIME);
  digitalWrite(PWM_A, 0);

  motorExtended = !motorExtended; // flip state
}
```

# Critical Reflection
The single pin testing is in this video: 
https://youtube.com/shorts/QNMFdwNKOtY

## Pin System Casing
The casing definitely improved in terms of size. The construction is currently more ergonomic and definitely will improve pins placement in the casing as well as flexibility in terms of other electronic placement. The square shape without railing does its work quite well however when the movement occurs it is a little wobbly. It might occur because of the friction therefore the support railing parts might bring more stability while the system is moving.
## Electronic Components
### Motor
The motor is powerful enough to push the top pin up and down without any problems or malfunctions. The motor speed on the other hand is a little too slow therefore other motors with different gear ratios will be tested. 
### Motor Controller
The L298N motor controller unfortunately happened to be obsolete (draws too much current to move the motor component) and it was replaced with modern TB6612FNG Motor Controller which performs its action correctly. 
### Button 
Button works well but only when the braille pin top attachment is pressed in the center. When pressed on the edge unfortunately the button doesn't click properly. Other important consideration is haptic click feeling which in the case of regular click button is not very noticeable. The button tactility while clicking has to be more recognizable.
